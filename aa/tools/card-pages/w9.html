from collections import deque

graph = {"A": ["B", "C"], "B": ["D"],
         "C": ["D", "E"], "D": ["E"], "E": []}
indegree = {u: 0 for u in graph}
for u in graph:
    for v in graph[u]:
        indegree[v] += 1
ready = deque(u for u in graph if indegree[u] == 0)
order = []
while ready:
    u = ready.popleft()
    order.append(u)
    for v in graph[u]:
        indegree[v] -= 1
        if indegree[v] == 0:
            ready.append(v)
if len(order) == len(graph):
    print("Valid order:", order)
else:
    print("Cycle detected. Partial order:", order)
