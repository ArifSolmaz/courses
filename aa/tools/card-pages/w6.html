import heapq
jobs = [5, 9, 7]
heapq.heapify(jobs)       # Make the list a min-heap.
heapq.heappush(jobs, 2)   # Add a new, earlier deadline.
print("Next:", jobs[0])
print("Removed:", heapq.heappop(jobs))
print("Remaining heap:", jobs)
