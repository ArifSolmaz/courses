Key Concepts: Quick Reference
Defining a Class & Creating Objects
# A class is a blueprint
class LED:
    pass

# Create two LED objects
red_led   = LED()
green_led = LED()

# Attach attributes with dot notation
red_led.color  = "red"
red_led.pin    = 13
red_led.is_on  = False

# Inspect
print(type(red_led))   # <class 'LED'>
print(vars(red_led))   # {'color': 'red', ...}
Objects Hold Independent State
class Motor:
    pass

m1 = Motor()
m2 = Motor()

m1.name  = "Left Wheel"
m1.speed = 120

m2.name  = "Right Wheel"
m2.speed = 95

# Each object has its own state
print(m1.speed)  # 120
print(m2.speed)  # 95
print(m1 is m2)  # False
