Key Concepts: Quick Reference
Variables & Types
# Declaring variables
name        = "Alice"   # str
age         = 20       # int
height      = 1.72     # float
is_enrolled = True     # bool

# Inspect types
print(type(age))      # <class 'int'>
print(type(height))   # <class 'float'>
print() Patterns
# Basic print
print("Hello, World!")

# With variable
temp_c = 37.5
print("Temperature:", temp_c)

# f-string (preview of Week 2)
print(f"Temp = {temp_c} °C")

# Calculation inside print
print(f"= {temp_c * 9/5 + 32} °F")
