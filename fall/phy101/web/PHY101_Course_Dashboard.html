Vector Components
# Decompose vector into components
import numpy as np

magnitude = 10    # N
angle     = 30    # degrees
theta     = np.radians(angle)

Fx = magnitude * np.cos(theta)
Fy = magnitude * np.sin(theta)
Vector Addition
# Add two vectors
A = np.array([3, 4])
B = np.array([1, 2])
R = A + B

# Magnitude of resultant
mag = np.linalg.norm(R)
print(f"|R| = {mag:.2f}")
